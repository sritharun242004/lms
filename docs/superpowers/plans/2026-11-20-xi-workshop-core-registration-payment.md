# XI Workshop Core: Registration, Demo Payment, Verification, Onboarding — Implementation Plan

> **For agentic workers:** This plan is executed inline in the current
> session (user explicitly waived subagent-driven-development's
> per-task review gate for this build). Steps use checkbox (`- [ ]`)
> syntax for tracking.

**Goal:** Build the registration → demo-QR payment → admin verification →
automatic participant onboarding path for the X Intelligence workshop
(Modules A–E of the master request), landing on a working, tested flow.

**Architecture:** Additive Prisma models (`WorkshopRegistration`,
`PaymentProof`, `PaymentVerificationLog`), a new `SUPER_ADMIN` role value, a
swappable demo QR provider, a transactional approve-and-onboard service
function, and three new route groups: a public marketing route for the
landing page + registration + payment UI, and an admin route for the
verification dashboard.

**Tech Stack:** Next.js 16 App Router, Prisma 6/Postgres, Zod, the existing
`qrcode` npm package (new dependency — pure JS, no native deps), Vitest.

**Spec:** `docs/superpowers/specs/2026-11-20-xi-workshop-platform-design.md`

## Global Constraints

- Pricing: Day 1 ₹28,250, Day 2 ₹31,275 (configurable — brief flags possible
  ₹31,276 rounding), Both days ₹55,490. All incl. 18% GST. One source of
  truth: `lib/workshop/pricing.ts`.
- No real payment credentials exist. The QR is a visible demo, never implying
  payment received. `PAYMENT_DEMO_MODE` env flag, defaults to `"true"`.
- Uploading a proof never auto-approves anything — approval is a distinct
  admin action.
- Reuse existing auth (`getCurrentUser`, `requireRole`), existing
  `errorResponse`/`successResponse` helpers, existing `AuditLog` model and
  `AuditAction` enum pattern, existing `prisma.$transaction` style (see
  `lib/admin/coach-management.ts`).
- Existing `UserRole` values `ADMIN`/`MENTOR`/`MENTEE` keep their current
  meaning. Only one enum value is added: `SUPER_ADMIN`.
- No change to any existing table's columns. No change to existing tests'
  expected behavior.
- Money is stored as integer paise-free rupees (`Int`, whole rupees — the
  brief's prices are all whole rupees) — never floats.

    const existing = await prisma.group.findFirst({ where: { name } });
    if (existing) continue;
    await prisma.group.create({
      data: { name, description: "X Intelligence workshop participants", createdById: superAdmin.id },
    });
  }
}

async function main() {
  await promoteBootstrapSuperAdmin();
  await ensureWorkshopGroups();
}

// apps/web/src/lib/workshop/pricing.ts

// Single source of truth for workshop package pricing, all figures incl.
// 18% GST per the approved brief. Day 2's base fee (₹26,505) times 1.18
// is ₹31,275.90 — the brief flags this rounding as unconfirmed, so it is
// kept here as a plain config value rather than computed, making the
// eventual business decision a one-line change.
export const WORKSHOP_PACKAGE_PRICING_INR = {
  DAY1: 28250,
  DAY2: 31275,
  BOTH: 55490,
} as const satisfies Record<"DAY1" | "DAY2" | "BOTH", number>;

export type WorkshopPackageKey = keyof typeof WORKSHOP_PACKAGE_PRICING_INR;

export function getPackageAmountInr(pkg: WorkshopPackageKey): number {
  return WORKSHOP_PACKAGE_PRICING_INR[pkg];
}
```

// apps/web/src/lib/workshop/registration.ts
import { z } from "zod";
import { prisma } from "@/lib/db/prisma";
import { getPackageAmountInr, type WorkshopPackageKey } from "./pricing";
import type { WorkshopRegistration } from "@prisma/client";

export class RegistrationError extends Error {
  constructor(message: string, readonly code: string, readonly status: number) {
    super(message);
    this.name = "RegistrationError";
  }
}

export const registrationInputSchema = z
  .object({
    fullName: z.string().trim().min(2).max(120),
    email: z.string().trim().toLowerCase().email(),
    whatsappNumber: z.string().trim().min(8).max(20),
    company: z.string().trim().min(1).max(160),
    designation: z.string().trim().min(1).max(120),
    workshopPackage: z.enum(["DAY1", "DAY2", "BOTH"]),
    previousDay1Attendance: z.boolean().optional(),
    previousDay1When: z.string().trim().max(200).optional(),
    aiExperienceLevel: z.enum(["NEW", "SOMETIMES", "DAILY"]),
    aiUseCase: z.string().trim().max(500).optional(),
    subscriptions: z.array(z.string()).default([]),
    needsGstInvoice: z.boolean().default(false),
    gstCompanyName: z.string().trim().max(200).optional(),
    gstin: z.string().trim().max(20).optional(),
    referredBy: z.string().trim().max(160).optional(),
    ageConfirmed: z.literal(true),
  })
  .superRefine((value, ctx) => {
    if (value.workshopPackage === "DAY2") {
      if (!value.previousDay1Attendance) {
        ctx.addIssue({
          code: z.ZodIssueCode.custom,
          path: ["previousDay1Attendance"],
          message: "Day 2 requires prior Day 1 attendance — choose Both days instead",
        });
      } else if (!value.previousDay1When) {
        ctx.addIssue({
          code: z.ZodIssueCode.custom,
          path: ["previousDay1When"],
          message: "Say when you completed Day 1",
        });
      }
    }
  });

export type RegistrationInput = z.infer<typeof registrationInputSchema>;

export function generateRegistrationCode(year: number, sequence: number): string {
  return `XI-${year}-${String(sequence).padStart(4, "0")}`;
}

async function nextSequenceForYear(year: number): Promise<number> {
  const count = await prisma.workshopRegistration.count({
    where: { registrationCode: { startsWith: `XI-${year}-` } },
  });
  return count + 1;
}

export async function createRegistration(
  input: RegistrationInput
): Promise<WorkshopRegistration> {
  const parsed = registrationInputSchema.safeParse(input);
  if (!parsed.success) {
    throw new RegistrationError(
      parsed.error.issues[0]?.message ?? "Invalid registration",
      "VALIDATION_ERROR",
      400
    );
  }
  const data = parsed.data;
  const amountInInr = getPackageAmountInr(data.workshopPackage as WorkshopPackageKey);
  const year = new Date().getFullYear();

  // Retry once on the unique-code race: two submissions in the same
  // millisecond could compute the same sequence number from `count()`.
  for (let attempt = 0; attempt < 2; attempt += 1) {
    const sequence = await nextSequenceForYear(year);
    try {
      return await prisma.workshopRegistration.create({
        data: {
          registrationCode: generateRegistrationCode(year, sequence),
          fullName: data.fullName,
          email: data.email,
          whatsappNumber: data.whatsappNumber,
          company: data.company,
          designation: data.designation,
          workshopPackage: data.workshopPackage,
          previousDay1Attendance: data.previousDay1Attendance,
          previousDay1When: data.previousDay1When,
          aiExperienceLevel: data.aiExperienceLevel,
          aiUseCase: data.aiUseCase,
          subscriptions: data.subscriptions,
          needsGstInvoice: data.needsGstInvoice,
          gstCompanyName: data.gstCompanyName,
          gstin: data.gstin,
          referredBy: data.referredBy,
          ageConfirmed: data.ageConfirmed,
          amountInInr,
          status: "PENDING_PAYMENT",
        },
      });
    } catch (error) {
      const isUniqueClash =
        typeof error === "object" && error && "code" in error && error.code === "P2002";
      if (!isUniqueClash || attempt === 1) throw error;
    }
  }
  throw new RegistrationError("Could not allocate a registration code", "CODE_CLASH", 500);
}

export async function findRegistrationByCode(
  registrationCode: string
): Promise<WorkshopRegistration | null> {
  return prisma.workshopRegistration.findUnique({ where: { registrationCode } });
}
```

import { NextRequest } from "next/server";
import { createRegistration, RegistrationError } from "@/lib/workshop/registration";
import { errorResponse, successResponse } from "@/lib/api/response";

export async function POST(req: NextRequest) {
  try {
    const body = await req.json();
    const registration = await createRegistration(body);
    return successResponse({ registration }, undefined, 201);
  } catch (error) {
    if (error instanceof RegistrationError) {
      return errorResponse(error.message, error.code, error.status);
    }
    return errorResponse("Unable to complete registration", "REGISTRATION_ERROR", 400);
  }
}

import QRCode from "qrcode";

export function isDemoPaymentMode(): boolean {
  return process.env.PAYMENT_DEMO_MODE !== "false";
}

// Swap point: a real integration replaces this function's body with a

// apps/web/src/lib/workshop/payment-proof.ts
import { prisma } from "@/lib/db/prisma";
import type { WorkshopRegistration } from "@prisma/client";

export const MAX_PROOF_SIZE_BYTES = 10 * 1024 * 1024; // 10 MB
export const ALLOWED_PROOF_MIME_TYPES = ["image/png", "image/jpeg", "application/pdf"];

export class PaymentProofError extends Error {
  constructor(message: string, readonly code: string, readonly status: number) {
    super(message);
    this.name = "PaymentProofError";
  }
}

export async function submitPaymentProof(
  registrationCode: string,
  file: { mimeType: string; size: number; data: Buffer }
): Promise<WorkshopRegistration> {
  if (!ALLOWED_PROOF_MIME_TYPES.includes(file.mimeType)) {
    throw new PaymentProofError(
      "Unsupported file type. Upload a PNG, JPG, or PDF.",
      "INVALID_FILE_TYPE",
      400
    );
  }
  if (file.size > MAX_PROOF_SIZE_BYTES) {
    throw new PaymentProofError("File is too large. Maximum size is 10 MB.", "FILE_TOO_LARGE", 400);
  }

  const registration = await prisma.workshopRegistration.findFirst({
    where: { registrationCode },
  });
  if (!registration) {
    throw new PaymentProofError("Registration not found", "NOT_FOUND", 404);
  }

  return prisma.$transaction(async (tx) => {
    await tx.paymentProof.create({
      data: {
        registrationId: registration.id,
        mimeType: file.mimeType,
        size: file.size,
        data: file.data,
      },
    });
    return tx.workshopRegistration.update({
      where: { id: registration.id },
      data: { status: "PROOF_SUBMITTED" },
    });
  });
}

export async function POST(
  req: NextRequest,
  { params }: { params: Promise<{ code: string }> }
) {
  const { code } = await params;
  const form = await req.formData();
  const file = form.get("file");
  if (!(file instanceof File)) return errorResponse("Choose a file to upload", "FILE_REQUIRED", 400);

  try {
    const buffer = Buffer.from(await file.arrayBuffer());
    const registration = await submitPaymentProof(code, {
      mimeType: file.type,
      size: file.size,
      data: buffer,
    });
    return successResponse({ registration }, undefined, 201);
  } catch (error) {
    if (error instanceof PaymentProofError) {
      return errorResponse(error.message, error.code, error.status);
    }
    return errorResponse("Unable to submit payment proof", "PAYMENT_PROOF_ERROR", 400);
  }

// apps/web/src/lib/workshop/verification.ts
import { prisma } from "@/lib/db/prisma";
import { hashPassword } from "@/lib/auth";
import { consoleNotificationService, type NotificationService } from "./notifications";
import crypto from "crypto";
import type { WorkshopRegistration } from "@prisma/client";

export class VerificationError extends Error {
  constructor(message: string, readonly code: string, readonly status: number) {
    super(message);
    this.name = "VerificationError";
  }
}

const GROUP_NAME_BY_PACKAGE: Record<string, string[]> = {
  DAY1: ["XI Workshop · Day 1"],
  DAY2: ["XI Workshop · Day 2"],
  BOTH: ["XI Workshop · Day 1", "XI Workshop · Day 2"],
};

async function requireProofSubmitted(registrationId: string) {
  const registration = await prisma.workshopRegistration.findUnique({
    where: { id: registrationId },
  });
  if (!registration) {
    throw new VerificationError("Registration not found", "NOT_FOUND", 404);
  }
  if (
    registration.status !== "PROOF_SUBMITTED" &&
    registration.status !== "UNDER_VERIFICATION"
  ) {
    throw new VerificationError(
      "This registration has no payment proof awaiting review",
      "NO_PROOF_SUBMITTED",
      409
    );
  }
  return registration;
}

export async function approveRegistration(
  adminId: string,
  registrationId: string,
  note?: string,
  notifications: NotificationService = consoleNotificationService
): Promise<WorkshopRegistration> {
  const registration = await requireProofSubmitted(registrationId);

  const updated = await prisma.$transaction(async (tx) => {
    let user = await tx.user.findFirst({ where: { email: registration.email } });
    if (!user) {
      user = await tx.user.create({
        data: {
          name: registration.fullName,
          email: registration.email,
          password: await hashPassword(crypto.randomBytes(24).toString("hex")),
          role: "MENTEE",
          emailVerified: true,
        },
      });
    }

    const groupNames = GROUP_NAME_BY_PACKAGE[registration.workshopPackage] ?? [];
    const groups = await tx.group.findMany({ where: { name: { in: groupNames } } });
    for (const group of groups) {
      await tx.groupMember.create({
        data: { userId: user.id, groupId: group.id, role: "MENTEE" },
      }).catch((error: unknown) => {
        const isDuplicate =
          typeof error === "object" && error && "code" in error && error.code === "P2002";
        if (!isDuplicate) throw error;
      });
    }

    const result = await tx.workshopRegistration.update({
      where: { id: registration.id },
      data: { status: "APPROVED", onboardedUserId: user.id },
    });

    await tx.paymentVerificationLog.create({
      data: { registrationId: registration.id, action: "APPROVED", note, adminId },
    });

    return result;
  });

  await notifications.sendApproval(updated);
  return updated;
}

export async function rejectRegistration(
  adminId: string,
  registrationId: string,
  note: string,
  notifications: NotificationService = consoleNotificationService
): Promise<WorkshopRegistration> {
  const registration = await requireProofSubmitted(registrationId);

  const updated = await prisma.$transaction(async (tx) => {
    const result = await tx.workshopRegistration.update({
      where: { id: registration.id },
      data: { status: "REJECTED" },
    });
    await tx.paymentVerificationLog.create({
      data: { registrationId: registration.id, action: "REJECTED", note, adminId },
    });
    return result;
  });

  await notifications.sendRejection(updated, note);
  return updated;
}

export async function requestResubmission(
  adminId: string,
  registrationId: string,
  note: string
): Promise<WorkshopRegistration> {
  const registration = await requireProofSubmitted(registrationId);

  return prisma.$transaction(async (tx) => {
    const result = await tx.workshopRegistration.update({
      where: { id: registration.id },
      data: { status: "RESUBMISSION_REQUESTED" },
    });
    await tx.paymentVerificationLog.create({
      data: {
        registrationId: registration.id,
        action: "RESUBMISSION_REQUESTED",
        note,
        adminId,
      },
    });
    return result;
  });
}

const ALLOWED_ROLES = new Set(["ADMIN", "SUPER_ADMIN"]);

export async function GET(req: NextRequest) {
  const user = await getCurrentUser();
  if (!user || !ALLOWED_ROLES.has(user.role)) {
    return errorResponse("Admin access required", "FORBIDDEN", 403);
  }

  const url = new URL(req.url);
  const status = url.searchParams.get("status");
  const q = url.searchParams.get("q");

  const registrations = await prisma.workshopRegistration.findMany({
    where: {
      ...(status ? { status: status as any } : {}),
      ...(q
        ? {
            OR: [
              { fullName: { contains: q, mode: "insensitive" } },
              { email: { contains: q, mode: "insensitive" } },
              { registrationCode: { contains: q, mode: "insensitive" } },
            ],
          }
        : {}),
    },
    orderBy: { createdAt: "desc" },
    include: { paymentProofs: { orderBy: { submittedAt: "desc" }, take: 1, select: { id: true, mimeType: true, submittedAt: true } } },
  });

  const grouped = await prisma.workshopRegistration.groupBy({
    by: ["status"],
    _count: true,
    _sum: { amountInInr: true },

  const summary = {
    total: grouped.reduce((sum, g) => sum + g._count, 0),
    byStatus: Object.fromEntries(grouped.map((g) => [g.status, g._count])),
    approvedRevenueInInr: grouped.find((g) => g.status === "APPROVED")?._sum.amountInInr ?? 0,

  const user = await getCurrentUser();
  if (!user || !ALLOWED_ROLES.has(user.role)) {
    return errorResponse("Admin access required", "FORBIDDEN", 403);
  }

  const { id } = await params;
  const body = await req.json();
  const note = typeof body.note === "string" ? body.note : undefined;

  try {
    if (body.action === "APPROVE") {
      const registration = await approveRegistration(user.id, id, note);
      return successResponse({ registration });
    }
    if (body.action === "REJECT") {
      const registration = await rejectRegistration(user.id, id, note ?? "Rejected");
      return successResponse({ registration });
    }
    if (body.action === "REQUEST_RESUBMISSION") {
      const registration = await requestResubmission(user.id, id, note ?? "Resubmission requested");
      return successResponse({ registration });
    }
    return errorResponse("Unknown action", "INVALID_ACTION", 400);
  } catch (error) {
    if (error instanceof VerificationError) {
      return errorResponse(error.message, error.code, error.status);
    }
    throw error;
  }
}

import { NextRequest, NextResponse } from "next/server";
import { prisma } from "@/lib/db/prisma";
import { getCurrentUser } from "@/lib/auth";
import { errorResponse } from "@/lib/api/response";

const ALLOWED_ROLES = new Set(["ADMIN", "SUPER_ADMIN"]);

export async function GET(
  _req: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const user = await getCurrentUser();
  if (!user || !ALLOWED_ROLES.has(user.role)) {
    return errorResponse("Admin access required", "FORBIDDEN", 403);
  }

  const { id } = await params;
  const proof = await prisma.paymentProof.findFirst({
    where: { registrationId: id },
    orderBy: { submittedAt: "desc" },
  });
  if (!proof) return errorResponse("No payment proof on file", "NOT_FOUND", 404);

  return new NextResponse(new Uint8Array(proof.data), {
    headers: {
      "Content-Type": proof.mimeType,
      "Content-Length": String(proof.size),
      "Content-Disposition": "inline",
      "Cache-Control": "private, no-store",
    },
  });
}
